cmake_minimum_required(VERSION 3.13)
if(NOT DEFINED ENV{PICO_SDK_PATH})
  message(FATAL_ERROR "Set PICO_SDK_PATH to the checked-out official Pico SDK")
endif()
include($ENV{PICO_SDK_PATH}/external/pico_sdk_import.cmake)
project(ball_bridge C CXX ASM)
set(CMAKE_C_STANDARD 11)
pico_sdk_init()
add_executable(ball_bridge core.c pico_main.c)
target_include_directories(ball_bridge PRIVATE ${CMAKE_CURRENT_LIST_DIR})
target_link_libraries(ball_bridge pico_stdlib pico_stdio_usb)
pico_enable_stdio_usb(ball_bridge 1)
pico_enable_stdio_uart(ball_bridge 0)
target_compile_definitions(ball_bridge PRIVATE PICO_STDIO_USB_CONNECT_WAIT_TIMEOUT_MS=0 PICO_STDIO_USB_STDOUT_TIMEOUT_US=1000)
pico_add_extra_outputs(ball_bridge)
