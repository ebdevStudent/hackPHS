73 def budget(self): 

74 counts = np.ceil((np.array(self.permitted.maximum) - self.permitted.minimum) / self.voxel_m) 

75 if np.prod(counts) > 100000: raise ValueError('Small-zone mapper limited to 100000 voxels')

80def merge_voxels(cells, origin, step, maximum): 

81 """Merge exact unions of adjacent voxel boxes; no filling unknown gaps.""" 

82 boxes = [(tuple(cell), tuple(i + 1 for i in cell)) for cell in sorted(cells)] 

83 for axis in (0, 1, 2): 

84 groups = {} 

85 for low, high in boxes: 

86 key = tuple((low[i], high[i]) for i in range(3) if i != axis) 

87 groups.setdefault(key, []).append((low, high)) 

88 boxes = [] 

89 for group in groups.values(): 

90 group.sort(key=lambda b: b[0][axis]) 

91 low, high = group[0] 

92 for other_low, other_high in group[1:]: 

93 if high[axis] == other_low[axis]: 

94 high = tuple(other_high[i] if i == axis else high[i] for i in range(3)) 

95 else: 

96 boxes.append((low, high)); low, high = other_low, other_high 

97 boxes.append((low, high)) 

98 return [Box(minimum=(origin + np.array(low) * step).tolist(), maximum=np.minimum(origin + np.array(high) * step, maximum).tolist()) for low, high in boxes]

101def build_map(scan, root): 

102 """Deterministic rebuild; validates hashes and all array data before mapping.""" 

103 scan = Scan.model_validate(scan.model_dump() if isinstance(scan, Scan) else scan) 

104 root = Path(root).resolve() 

105 loaded = [] 

106 for frame in scan.frames: 

107 path = (root / frame.file).resolve() 

108 if path.parent != root: raise ValueError('Depth path escapes capture directory') 108 ↛ exitline 108 didn't except from function 'build_map' because the raise on line 108 wasn't executed

109 if path.stat().st_size > 30_000_000: raise ValueError('Depth file too large') 109 ↛ exitline 109 didn't except from function 'build_map' because the raise on line 109 wasn't executed

110 if hashlib.sha256(path.read_bytes()).hexdigest() != frame.sha256: raise ValueError('Depth hash mismatch') 

111 with np.load(path, allow_pickle=False) as archive: 

112 if 'depth_m' not in archive: raise ValueError('Missing metric depth array') 112 ↛ exitline 112 didn't except from function 'build_map' because the raise on line 112 wasn't executed

113 depth = archive['depth_m'] 

114 if depth.shape != (frame.intrinsics.height, frame.intrinsics.width) or depth.dtype.kind != 'f' or not np.isfinite(depth).all() or np.any(depth < 0): raise ValueError('Invalid depth units/shape/nonfinite samples') 114 ↛ exitline 114 didn't except from function 'build_map' because the raise on line 114 wasn't executed

115 loaded.append((frame, depth.copy())) 

116 origin, maximum = np.array(scan.permitted.minimum), np.array(scan.permitted.maximum) 

117 dims = np.ceil((maximum - origin) / scan.voxel_m).astype(int) 

118 occupied, free = set(), set() 

119 for frame, depth in loaded: 

120 k, transform = frame.intrinsics, np.array(frame.map_from_depth) 

121 y, x = np.indices(depth.shape) 

122 valid = (depth > 0) & (depth <= scan.max_depth_m) 

123 z = depth[valid] 

124 points = np.column_stack(((x[valid] - k.cx) * z / k.fx, (y[valid] - k.cy) * z / k.fy, z)) 

125 world = points @ transform[:3, :3].T + transform[:3, 3] 

126 indices = np.floor((world - origin) / scan.voxel_m).astype(int) 

127 for index, point in zip(indices, world): 

128 if np.all(index >= 0) and np.all(index < dims) and np.all(point < maximum): occupied.add(tuple(index)) 

129 camera_from_map = invert(transform) 

130 for index in itertools.product(*(range(int(d)) for d in dims)): 

131 if index in occupied: continue 

132 lo = origin + np.array(index) * scan.voxel_m 

133 hi = np.minimum(lo + scan.voxel_m, maximum) 

134 corners = np.array(list(itertools.product(*zip(lo, hi)))) 

135 camera = corners @ camera_from_map[:3, :3].T + camera_from_map[:3, 3] 

136 if np.any(camera[:, 2] <= .1): continue 

137 pixels = camera[:, :2] / camera[:, 2, None] * [k.fx, k.fy] + [k.cx, k.cy] 

138 lower, upper = np.floor(pixels.min(axis=0)).astype(int), np.ceil(pixels.max(axis=0)).astype(int) 

139 if np.any(lower < 0) or np.any(upper >= [k.width, k.height]): continue 

140 patch = depth[lower[1]:upper[1] + 1, lower[0]:upper[0] + 1] 

141 if np.all(patch > 0) and np.all(patch <= scan.max_depth_m) and float(patch.min()) > float(camera[:, 2].max()) + scan.uncertainty_m: 

142 free.add(index) 

143 free.difference_update(occupied) 

144 if not free: raise ValueError('No fully observed free volume; add registered station/coverage') 

145 free_boxes = merge_voxels(free, origin, scan.voxel_m, maximum) 

146 occupied_boxes = merge_voxels(occupied, origin, scan.voxel_m, maximum) 

147 if len(free_boxes) > 1000 or len(occupied_boxes) > 1000: raise ValueError('Map fragmentation exceeds contract: reduce zone or inspect scan quality') 147 ↛ exitline 147 didn't except from function 'build_map' because the raise on line 147 wasn't executed

148 m = Map(schema_version=1, map_id=scan.map_id, map_version=scan.map_version, frame_id=scan.frame_id, axes=scan.axes, units='m', permitted=scan.permitted, 

149 observed_free=free_boxes, occupied=occupied_boxes, uncertainty_m=scan.uncertainty_m, acquired_utc=scan.acquired_utc, evidence=scan.evidence, example=scan.example) 

150 digest = hashlib.sha256(scan.model_dump_json().encode()).hexdigest() 

151 report = {'scan_id': scan.scan_id, 'scan_sha256': digest, 'map_sha256': hashlib.sha256(m.model_dump_json().encode()).hexdigest(), 

152 'voxel_m': scan.voxel_m, 'occupied_voxels': len(occupied), 'observed_free_voxels': len(free), 'unknown_voxels': int(np.prod(dims)) - len(free) - len(occupied), 

153 'frames': len(loaded), 'physical_acceptance': False, 'example': scan.example, 

154 'limitations': 'Static surveyed stations; no SLAM, dynamic obstacle sensing or glass/reflectivity guarantee. Recheck controlled zone before use.'} 

155 return m, report

158def rebuild_scans(paths): 

159 """Combine independently registered stationary captures without changing geometry.""" 

160 import shutil 

161 import tempfile 

162 if not paths or len(paths) > 20: raise ValueError('Supply 1–20 registered scans') 

163 scans = [(Path(path), Scan.model_validate_json(Path(path).read_text())) for path in paths] 

164 first = scans[0][1] 

165 fields = ('map_id', 'map_version', 'frame_id', 'units', 'axes', 'permitted', 'voxel_m', 'uncertainty_m', 'max_depth_m', 'example') 

166 for _, scan in scans: 

167 if any(getattr(first, field) != getattr(scan, field) for field in fields): raise ValueError('Registered scan map/settings mismatch') 

168 with tempfile.TemporaryDirectory(prefix='ball-map-rebuild-') as directory: 

169 combined = [] 

170 for index, (path, scan) in enumerate(scans): 

171 root = path.parent.resolve() 

172 for frame in scan.frames: 

173 source = (root / frame.file).resolve() 

174 if source.parent != root or source.stat().st_size > 30_000_000: raise ValueError('Invalid capture file path/size') 174 ↛ exitline 174 didn't except from function 'rebuild_scans' because the raise on line 174 wasn't executed

175 name = f'station-{index}-{frame.file}' 

176 shutil.copyfile(source, Path(directory) / name) 

177 combined.append(frame.model_copy(update={'file': name})) 

178 data = first.model_dump() 

179 data.update(frames=[frame.model_dump() for frame in combined], evidence='; '.join(scan.evidence for _, scan in scans)) 

180 mapped, report = build_map(Scan.model_validate(data), directory) 

181 report['source_manifests'] = [{'path': str(path.resolve()), 'sha256': hashlib.sha256(path.read_bytes()).hexdigest()} for path, _ in scans] 

182 return mapped, report

205def capture_d435(station, output, frames=30, sdk=None): 

206 """Native rectified depth + color + raw BAG; no alignment/exposure-time guesses.""" 

207 import importlib.metadata 

208 import time 

209 from datetime import datetime, timezone 

210 if type(frames) is not int or not 1 <= frames <= 1000: raise ValueError('Capture 1–1000 frames') 

211 station = Station.model_validate(station.model_dump() if isinstance(station, Station) else station) 

212 if station.example: raise ValueError('Measure station registration; examples cannot capture real hardware') 

213 if sdk is None: 

214 try: import pyrealsense2 as sdk 

215 except ImportError as error: raise ValueError('D435 SDK unavailable on this Python/platform; use a supported capture environment and import saved frames') from error 

216 output = Path(output) 

217 output.mkdir(parents=True, exist_ok=False) 

218 pipeline, config = sdk.pipeline(), sdk.config() 

219 config.enable_stream(sdk.stream.depth, 640, 480, sdk.format.z16, 30) 

220 config.enable_stream(sdk.stream.color, 640, 480, sdk.format.bgr8, 30) 

221 config.enable_record_to_file(str(output / 'raw.bag')) 

222 started = False 

223 try: 

224 profile = pipeline.start(config) 

225 started = True 

226 device = profile.get_device() 

227 name = device.get_info(sdk.camera_info.name) 

228 if name.strip() not in ('Intel RealSense D435', 'Intel RealSense D435 Camera', 'RealSense D435'): raise ValueError('Expected exact D435 device; review capabilities for other models') 

229 scale = device.first_depth_sensor().get_depth_scale() 

230 if not np.isfinite(scale) or scale <= 0: raise ValueError('Invalid SDK depth scale') 

231 for _ in range(30): pipeline.wait_for_frames(5000) 

232 captures, metadata = [], {} 

233 for i in range(frames): 

234 sample = pipeline.wait_for_frames(5000) 

235 received = time.monotonic() 

236 depth, color = sample.get_depth_frame(), sample.get_color_frame() 

237 if not depth or not color: raise ValueError('Incomplete D435 frame set') 

238 dprofile, cprofile = depth.profile.as_video_stream_profile(), color.profile.as_video_stream_profile() 

239 intr = dprofile.get_intrinsics() 

240 if any(abs(c) > 1e-10 for c in intr.coeffs): raise ValueError('Mapper requires rectified pinhole depth; preserve raw BAG and explicitly rectify') 

241 k = DepthIntrinsics(width=intr.width, height=intr.height, fx=float(intr.fx), fy=float(intr.fy), cx=float(intr.ppx), cy=float(intr.ppy), model='rectified_pinhole') 

242 raw_depth = np.asanyarray(depth.get_data()) 

243 if raw_depth.dtype != np.uint16: raise ValueError('Expected z16 depth format') 

244 filename = f'frame-{i:04d}.npz' 

245 np.savez_compressed(output / filename, depth_m=raw_depth.astype(float) * scale, raw_depth=raw_depth, color_bgr=np.asanyarray(color.get_data())) 

246 captures.append(DepthFrame(file=filename, sha256=hashlib.sha256((output / filename).read_bytes()).hexdigest(), intrinsics=k, 

247 map_from_depth=station.map_from_depth, registration_evidence=station.registration_evidence, 

248 hardware_timestamp_ms=float(depth.get_timestamp()), hardware_clock_domain=str(depth.get_frame_timestamp_domain()), host_received_s=received)) 

249 ci = cprofile.get_intrinsics() 

250 extr = dprofile.get_extrinsics_to(cprofile) 

251 metadata = {'device_name': name, 'firmware': device.get_info(sdk.camera_info.firmware_version), 'depth_scale_m': scale, 

252 'depth_model': str(intr.model), 'depth_coefficients': list(intr.coeffs), 

253 'color_intrinsics': {'width': ci.width, 'height': ci.height, 'fx': ci.fx, 'fy': ci.fy, 'cx': ci.ppx, 'cy': ci.ppy, 'model': str(ci.model), 'coefficients': list(ci.coeffs)}, 

254 'depth_to_color': {'rotation_column_major': list(extr.rotation), 'translation_m': list(extr.translation)}, 

255 'color_hardware_timestamp_ms': float(color.get_timestamp()), 'color_clock_domain': str(color.get_frame_timestamp_domain()), 

256 'timestamp_note': 'Device clock timestamps and host receipt remain separate; no host exposure timestamp inferred.'} 

257 values = station.model_dump(exclude={'map_from_depth', 'registration_evidence'}) 

258 values.update(camera_model='D435', serial=device.get_info(sdk.camera_info.serial_number), sdk_version=importlib.metadata.version('pyrealsense2') if sdk.__name__ == 'pyrealsense2' else 'injected-test-sdk', 

259 units='m', axes='right_handed_z_up', acquired_utc=datetime.now(timezone.utc).isoformat(), evidence=station.registration_evidence, frames=captures, camera_metadata=metadata) 

260 scan = Scan.model_validate(values) 

261 (output / 'scan.json').write_text(scan.model_dump_json(indent=2) + '\n') 

262 return scan 

263 finally: 

264 if started: pipeline.stop()
